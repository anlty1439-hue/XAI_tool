/* App state */
const state = {
  route: "home",
  currentPatientId: null,
  currentMethod: null, // "shap" | "lime"
  currentGraph: null   // { key, title, img }
};

/* Utilities */
function showRoute(name){
  state.route = name;
  document.querySelectorAll(".route").forEach(s=>{
    s.classList.toggle("active", s.dataset.name===name);
  });
  if(name==="patients") renderPatients();
  if(name==="patient") renderPatient();
  if(name==="viz-options") renderVizOptions();
  if(name==="graph") renderGraph();
  window.location.hash = "#/"+name;
}

function toastMsg(text, ms=2800){
  toast.textContent = text;
  toast.classList.add("show");
  setTimeout(()=> toast.classList.remove("show"), ms);
}

/* Initialize routes from hash */
function initFromHash(){
  const h = window.location.hash.replace("#/","");
  if(routes.includes(h)) showRoute(h);
  else showRoute("home");
}

/* Renderers */
function renderPatients(){
  const body = document.getElementById("patients-body");
  body.innerHTML = "";
  patients.forEach(p=>{
    const tr = document.createElement("tr");
    tr.innerHTML = `
      <td>${p.id}</td>
      <td>${p.level}</td>
      <td>
        ${p.available
          ? `<span class="${p.consent ? 'label-danger' : 'label-ok'}">
               Available
             </span>`
          : `<span class="label-na">Not available</span>`
        }
      </td>
      <td>
        <button class="btn"
          ${!p.available ? 'disabled' : ''}
          data-open-patient="${p.id}">
          ${p.available ? 'Open' : '—'}
        </button>
      </td>
    `;
    body.appendChild(tr);
  });
}

function levelToExample(level){
  // Map to the closest example group for feature display
  if(level==="Obese") return "Obese";
  if(level==="Overweight") return "Overweight";
  if(level==="Underweight") return "Underweight";
  return "Normal";
}

function renderPatient(){
  const p = patients.find(x=> x.id===state.currentPatientId);
  const subtitle = document.getElementById("patient-subtitle");
  subtitle.textContent = p ? `ID ${p.id} • Weight level: ${p.level}` : "";

  const tbody = document.getElementById("patient-features");
  tbody.innerHTML = "";
  if(!p) return;

  const exampleKey = levelToExample(p.level);
  const rows = featureSets[exampleKey] || [];
  rows.forEach(([k,v])=>{
    const tr = document.createElement("tr");
    tr.innerHTML = `<td>${k}</td><td>${v}</td>`;
    tbody.appendChild(tr);
  });
}

function renderVizOptions(){
  const grid = document.getElementById("viz-grid");
  const title = document.getElementById("viz-method-title");
  const list = vizByMethod[state.currentMethod] || [];
  title.textContent = state.currentMethod==="shap"
    ? "SHAP visualisation options"
    : "LIME visualisation options";
  grid.innerHTML = "";
  list.forEach(item=>{
    const card = document.createElement("div");
    card.className = "viz-card";
    card.innerHTML = `
      <div class="viz-thumb">
        <img src="${item.img}" alt="${item.title}" style="max-height:120px; width:auto" />
      </div>
      <div class="viz-title">${item.title}</div>
      <button class="btn btn-primary" data-open-graph="${item.key}">Open</button>
    `;
    grid.appendChild(card);
  });
}

function renderGraph(){
  const title = document.getElementById("graph-title");
  const img = document.getElementById("graph-image");
  if(!state.currentGraph) return;
  title.textContent = state.currentGraph.title;
  img.src = state.currentGraph.img;
  img.alt = state.currentGraph.title;
}

/* Modals / drawers */
const consentModal = document.getElementById("consent-modal");
const consentYes = document.getElementById("consent-yes");
let pendingOpenPatientId = null;

function openConsent(id){
  pendingOpenPatientId = id;
  consentModal.classList.add("show");
  consentModal.setAttribute("aria-hidden","false");
}
function closeConsent(){
  consentModal.classList.remove("show");
  consentModal.setAttribute("aria-hidden","true");
  pendingOpenPatientId = null;
}
document.querySelector('[data-close="consent"]')?.addEventListener("click", closeConsent);
consentYes.addEventListener("click", ()=>{
  const id = pendingOpenPatientId;
  closeConsent();
  if(id!=null){
    state.currentPatientId = id;
    showRoute("patient");
  }
});

/* Features modal */
const featuresModal = document.getElementById("features-modal");
function openFeatures(){
  const p = patients.find(x=> x.id===state.currentPatientId);
  const list = document.getElementById("features-list");
  list.innerHTML = "";
  if(!p) return;
  const exampleKey = levelToExample(p.level);
  (featureSets[exampleKey]||[]).forEach(([k,v])=>{
    const row = document.createElement("div");
    row.className = "row";
    row.innerHTML = `<div class="k">${k}</div><div class="v">${v}</div>`;
    list.appendChild(row);
  });
  featuresModal.classList.add("show");
  featuresModal.setAttribute("aria-hidden","false");
}
function closeFeatures(){
  featuresModal.classList.remove("show");
  featuresModal.setAttribute("aria-hidden","true");
}
document.querySelector('[data-close="features"]')?.addEventListener("click", closeFeatures);

/* Describe drawer */
const drawer = document.getElementById("describe-drawer");
const drawerBody = document.getElementById("describe-body");
document.getElementById("close-drawer").addEventListener("click", ()=> {
  drawer.classList.remove("open");
  drawer.setAttribute("aria-hidden","true");

function openDescribe(){
  const key = state.currentMethod+":"+state.currentGraph.key;
  drawerBody.textContent = (graphDescriptions[key]||"").trim();
  drawer.classList.add("open");
  drawer.setAttribute("aria-hidden","false");
}

/* Click handlers and navigation */
document.body.addEventListener("click",(e)=>{
  const btn = e.target.closest("button");
  if(!btn) return;

  // Top/inline route buttons
  if(btn.dataset.route){
    const route = btn.dataset.route;
    if(route==="viz-options" && btn.dataset.method){
      state.currentMethod = btn.dataset.method; // shap | lime
    }
    showRoute(route);

    // Instructional toasts
    if(btn.dataset.instruction==="patientsIntro"){
      toastMsg('Instruction: "Not available" means no details yet. Red "Available" requires consent.');
    }
    if(btn.dataset.instruction==="vizIcons"){
      toastMsg("Instruction: The graph icons on the next screen are clickable.");
    }
    return;
  }

  // Open patient buttons
  if(btn.dataset.openPatient){
    const id = Number(btn.dataset.openPatient);
    const p = patients.find(x=> x.id===id);
    if(!p) return;
    if(!p.available){
      toastMsg("This patient data is not available.");
      return;
    }
    if(p.consent){
      openConsent(id);
    } else {
      state.currentPatientId = id;
      showRoute("patient");
    }
    return;
  }

  // Open graph from options
  if(btn.dataset.openGraph){
    const key = btn.dataset.openGraph;
    const chosen = (vizByMethod[state.currentMethod]||[]).find(v=> v.key===key);
    state.currentGraph = chosen || null;
    showRoute("graph");
    return;
  }

  // Graph page actions
  if(btn.id==="btn-feature-details"){
    openFeatures();
    return;
  }
  if(btn.id==="btn-describe-graph"){
    openDescribe();
    return;
  }
  if(btn.id==="btn-print"){
    // Mirror "Sorry printing is not available" note from prototype for some pages,
    // but we’ll let the browser print the current graph area for practicality.
    window.print();
    return;
  }
  if(btn.id==="btn-other-graphs"){
    showRoute("viz-options");
    return;
  }
});
